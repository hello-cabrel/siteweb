  <script src="https://cdnjs.cloudflare.com/ajax/libs/gsap/3.12.2/ScrollTrigger.min.js"></script>

  <script>
    window.addEventListener('DOMContentLoaded', () => {
      const currentYearElement = document.getElementById('adg-current-year');
      if (currentYearElement) {
        currentYearElement.textContent = new Date().getFullYear();
      }

      const wrapper = document.querySelector('.tiramisu-wrapper');
      if (!wrapper) return;

      function setFooterReveal() {
        const footer = wrapper.querySelector('.adg-footer');
        const mainWrap = wrapper.querySelector('.adg-main-wrapper');
        if(footer && mainWrap) { mainWrap.style.marginBottom = footer.offsetHeight + 'px'; }
      }
      setTimeout(setFooterReveal, 300);
      window.addEventListener('resize', setFooterReveal);

      window.addEventListener('scroll', () => {
        const header = wrapper.querySelector('#adg-main-header');
        if (!header) return;
        if (window.scrollY > 50) { header.classList.add('adg-scrolled'); }
        else { header.classList.remove('adg-scrolled'); }
      });

      const hamburger = wrapper.querySelector('#adg-hamburger-menu');
      const navMenu = wrapper.querySelector('#adg-nav-menu');
      const header = wrapper.querySelector('#adg-main-header');

      if (hamburger && navMenu && header) {
        hamburger.addEventListener('click', () => {
          hamburger.classList.toggle('adg-active');
          navMenu.classList.toggle('adg-active');
          header.classList.toggle('adg-menu-open');
        });

        wrapper.querySelectorAll('.adg-nav-item').forEach(link => {
          link.addEventListener('click', () => {
            hamburger.classList.remove('adg-active');
            navMenu.classList.remove('adg-active');
            header.classList.remove('adg-menu-open');
          });
        });
      }

      if (typeof gsap !== 'undefined' && typeof ScrollTrigger !== 'undefined') {
        gsap.registerPlugin(ScrollTrigger);

        gsap.to(".tiramisu-wrapper .adg-gs-parallax", {
          yPercent: 30, ease: "none",
          scrollTrigger: { trigger: ".tiramisu-wrapper .adg-hero", start: "top top", end: "bottom top", scrub: true }
        });

        wrapper.querySelectorAll('.adg-gs-reveal').forEach((elem) => {
          gsap.fromTo(elem,
            { autoAlpha: 0, y: 50 },
            { duration: 1, autoAlpha: 1, y: 0, ease: "power3.out", scrollTrigger: { trigger: elem, start: "top 85%" } }
          );
        });

        gsap.to(".tiramisu-wrapper .adg-portal-bg", {
          yPercent: 10, ease: "none",
          scrollTrigger: { trigger: ".tiramisu-wrapper .adg-bento-grid", start: "top bottom", end: "bottom top", scrub: true }
        });
      }
    });
  </script>
</section>
